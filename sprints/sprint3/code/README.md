# โค้ด ณ วันจบ Sprint 3 (v0.3.1)

สำเนาโค้ดตอนจบ Sprint 3 เก็บไว้ดูย้อนหลัง โค้ดปัจจุบันอยู่ที่โฟลเดอร์หลักของโปรเจกต์

```bash
cd sprints/sprint3/code
pip install -r requirements.txt
python -m src.main       # เล่นผ่าน CLI
python -m web.app        # เล่นผ่านเว็บ http://127.0.0.1:5000
python -m pytest -q      # 58 passed
```
